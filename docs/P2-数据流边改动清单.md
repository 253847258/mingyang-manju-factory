### 2.2 端口校验（新增；`createLink` 内**追加**调用，不改其现有行为）

> **关键兼容点**：历史语义连线（对手戏/亲属）**没有端口**，必须继续可用。故校验**仅在显式传了 `from_port`/`to_port` 时生效**：

```js
function validatePorts({ canvas_id, from_node, to_node, from_port, to_port }) {
  if (!from_port && !to_port) return null;      // 语义连线：跳过端口校验，保持旧行为
  if (!from_port || !to_port) throw new Error('端口信息不完整（from_port / to_port 需同时提供）');
  const a = get('SELECT kind FROM canvas_nodes WHERE id = ?', from_node);
  const b = get('SELECT kind FROM canvas_nodes WHERE id = ?', to_node);
  const ka = a?.kind || 'actor'; const kb = b?.kind || 'actor';
  if (!(PORT_RULES[ka]?.out || []).includes(from_port)) throw new Error(`「${ka}」节点没有输出端口 ${from_port}`);
  if (!(PORT_RULES[kb]?.in || []).includes(to_port)) throw new Error(`「${kb}」节点没有输入端口 ${to_port}`);
  if (!portOk(from_port, to_port)) throw new Error(`不允许 ${from_port} → ${to_port} 的连接`);
  return { ka, kb };
}
```

### 6.2 `portPoint` 扩展为"端口序号感知"（**不删旧函数**，新增包装）

**定位锚点**：`function portPoint(it, side, inset = 0) {`

在其后新增（旧函数与其 7 处调用**一律不动**，避免回归）：

```js
/** 第 index 个（共 count 个）端口在该侧的位置；count<=1 时退化为原 portPoint 的中点 */
function portPointAt(it, side, index = 0, count = 1, inset = 0) {
  if (!it) return { x: 0, y: 0 };
  const r = Math.max(4, 6 / S.zoom) + inset;
  const f = count <= 1 ? 0.5 : (index + 1) / (count + 1);   // 沿侧边均分，避开端点
  if (side === 'left')  return { x: it.x - r,          y: it.y + it.h * f };
  if (side === 'right') return { x: it.x + it.w + r,   y: it.y + it.h * f };
  if (side === 'top')   return { x: it.x + it.w * f,   y: it.y - r };
  return { x: it.x + it.w * f, y: it.y + it.h + r };
}

/** 解析一条边在节点上的锚点：优先按端口名，回退到旧 side 语义 */
function edgeAnchor(it, portName, side, fallbackIndex = 0) {
  if (!it) return { x: 0, y: 0 };
  const p = portsOf(it);
  // 输出端口在右侧，输入端口在左侧（与 LibTV 的数据流方向一致）
  if (portName && p.out.includes(portName)) {
    return portPointAt(it, 'right', p.out.indexOf(portName), p.out.length);
  }
  if (portName && p.in.includes(portName)) {
    const ins = p.in; const idx = ins.indexOf(portName);
    return portPointAt(it, 'left', idx, ins.length);
  }
  return portPoint(it, side || 'right', 0);    // ★ 历史无端口连线：完全走旧逻辑
}
```
